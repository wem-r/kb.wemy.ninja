{{- /* ```terminal fences — a shell session in a window frame. ```terminal ```terminal {title="debian — ssh"} Rendered through OINK's own code pipeline (title bar, copy button, theme colours), with two additions: - prompts are recognised in their full form, which Chroma's `console` lexer does not do: user@host:~/dir$ root@host:/# [user@host dir]$ $ # % ❯ PS C:\path> C:\path> The prompt is marked as a Chroma prompt token (.gp), the command is highlighted as bash, every other line is output (.go). OINK's copy button therefore copies ONLY the commands, without prompts or output. - a `td-code--terminal` class for the window frame (assets/scss/_styles_project.scss). With lineNos / hl_lines options, OINK's plain `console` highlighting is used. */ -}} {{- $attrs := .Attributes | default dict -}} {{- $class := trim (printf "td-code--terminal %v" (index $attrs "class" | default "")) " " -}} {{- $attrs = merge $attrs (dict "class" $class) -}} {{- $config := partial "code/normalize.html" (dict "page" .Page "position" .Position "code" .Inner "language" "console" "attributes" $attrs "options" .Options "key" (printf "fence-%d" .Ordinal) ) -}} {{- $highlighted := "" -}} {{- if .Options -}} {{- $highlighted = partial "code/highlight.html" $config -}} {{- else -}} {{- $promptRE := `^((?:\[[^\]]+\]|[\w.-]+@[\w.-]+(?::[^\s$#%]*)?)\s?[$#%]|[$#%❯]|PS [^>]*>|[A-Za-z]:\\[^>]*>)( +)(.*)$` -}} {{- $out := slice -}} {{- range split (strings.TrimRight "\n" .Inner) "\n" -}} {{- $m := findRESubmatch $promptRE . 1 -}} {{- if $m -}} {{- $p := index $m 0 -}} {{- $cmd := "" -}} {{- with index $p 3 -}} {{- $cmd = transform.Highlight . "bash" "hl_inline=true,noClasses=false" | string -}} {{- $cmd = replaceRE `^]*>|$` "" $cmd -}} {{- end -}} {{- $out = $out | append (printf `%s%s%s ` (htmlEscape (index $p 1)) (index $p 2) $cmd) -}} {{- else -}} {{- $out = $out | append (printf `%s ` (htmlEscape .)) -}} {{- end -}} {{- end -}} {{- $highlighted = printf `
%s
` (delimit $out "") | safeHTML -}} {{- end -}} {{- partial "code/render.html" (dict "config" $config "highlighted" $highlighted) -}}
