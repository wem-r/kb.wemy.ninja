{{- /* `````terminal → terminal window, no title ````terminal {title="debian — ssh"} → title in the window bar Lines that start with a shell prompt get the prompt in mint and the command syntax-highlighted as bash; every other line is shown as plain output. Recognised prompts: user@host$ user@host:~/dir# [user@host dir]$ $ # PS C:\path> C:\path> ❯ */ -}} {{- $promptRE := `^((?:\[[^\]]+\]|[\w.-]+@[\w.-]+(?::[^\s$#%]*)?)\s?[$#%]|[$#%❯]|PS [^>]*>|[A-Za-z]:\\[^>]*>)( +)(.*)$` -}} {{- $title := .Attributes.title | default "" -}}
{{- with $title }}{{ . }}{{ end }} 

    {{- range $i, $line := split (strings.TrimRight "\n" .Inner) "\n" -}}
      {{- if $i }}{{ "\n" }}{{ end -}}
      {{- $m := findRESubmatch $promptRE $line 1 -}}
      {{- if $m -}}
        {{- $parts := index $m 0 -}}
        {{ index $parts 1 }}{{ index $parts 2 -}}
        {{ with index $parts 3 }}{{ transform.Highlight . "bash" "hl_inline=true" }}{{ end }}
      {{- else -}}
        {{ $line }}
      {{- end -}}
    {{- end -}}
